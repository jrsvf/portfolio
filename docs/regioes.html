Regiões de Desenvolvimento

As 12 Regiões de Desenvolvimento de Pernambuco: produção, emprego, cadeias, empresas, qualificação, ICMS e agricultura familiar.
Código 
regs = FileAttachment("dados/regioes.json").json()
geo0 = FileAttachment("dados/malha_pe.geojson").json()
geo = ({type: "FeatureCollection", features: geo0.features.map(f => {
  if (d3.geoArea(f) <= 2 * Math.PI) return f;
  const g = f.geometry;
  const rev = poly => poly.map(r => r.slice().reverse());
  const coords = g.type === "Polygon" ? rev(g.coordinates) : g.coordinates.map(rev);
  return {...f, geometry: {...g, coordinates: coords}};
})})
import {cartoes, blocoEmpresas, blocoQualif, blocoEducacao, blocoIcms, blocoAgro, blocoInfra} from "./territorio.js"
nomesR = regs.map(r => r.rd).sort((a, b) => a.localeCompare(b, "pt-BR"))
inicialR = {
  const q = new URLSearchParams(window.location.search).get("r");
  return q && nomesR.includes(q) ? q : "Metropolitana";
}
viewof rsel = Inputs.select(nomesR, {label: "Região de Desenvolvimento", value: inicialR})
r = {
  try { history.replaceState(null, "", "?r=" + encodeURIComponent(rsel)); } catch (e) {}
  return regs.find(x => x.rd === rsel);
}
ids = new Set(r.municipios.map(d => d.id))
nf = (x, d = 0) => x == null || isNaN(x) ? "–" : x.toLocaleString("pt-BR", {maximumFractionDigits: d, minimumFractionDigits: d})
pct = (x, d = 1) => x == null || isNaN(x) ? "–" : (100 * x).toLocaleString("pt-BR", {maximumFractionDigits: d, minimumFractionDigits: d}) + "%"
reais = x => x == null ? "–" : x >= 1e9 ? "R$ " + nf(x / 1e9, 2) + " bi" : x >= 1e6 ? "R$ " + nf(x / 1e6, 1) + " mi" : "R$ " + nf(x / 1e3, 0) + " mil"
AZUL = "#1F4E79"; LAR = "#E07A1F"
