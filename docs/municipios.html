Raio-X dos municípios

Produção, emprego, cadeias produtivas e comércio exterior de cada um dos 185 municípios de Pernambuco, com dados públicos.
Código 
dados = FileAttachment("dados/raiox_municipios.json").json()
geo0 = FileAttachment("dados/malha_pe.geojson").json()
// o d3 espera anéis no sentido horário; inverte os que vierem ao contrário
geo = ({type: "FeatureCollection", features: geo0.features.map(f => {
  if (d3.geoArea(f) <= 2 * Math.PI) return f;
  const g = f.geometry;
  const rev = poly => poly.map(r => r.slice().reverse());
  const coords = g.type === "Polygon" ? rev(g.coordinates) : g.coordinates.map(rev);
  return {...f, geometry: {...g, coordinates: coords}};
})})
nomes = dados.map(d => d.nome).sort((a, b) => a.localeCompare(b, "pt-BR"))
inicial = {
  const q = new URLSearchParams(window.location.search).get("m");
  const alvo = q ? dados.find(d => d.id === q || d.nome.toLowerCase() === q.toLowerCase()) : null;
  return alvo ? alvo.nome : "Recife";
}
viewof escolha = Inputs.select(nomes, {label: "Município", value: inicial})
m = dados.find(d => d.nome === escolha)

// formatação
nf = (x, d = 0) => x == null || isNaN(x) ? "–" : x.toLocaleString("pt-BR", {maximumFractionDigits: d, minimumFractionDigits: d})
pct = (x, d = 1) => x == null || isNaN(x) ? "–" : (x > 0 ? "+" : "") + (100 * x).toLocaleString("pt-BR", {maximumFractionDigits: d, minimumFractionDigits: d}) + "%"
reais = x => x == null ? "–" : x >= 1e9 ? "R$ " + nf(x / 1e9, 2) + " bi" : x >= 1e6 ? "R$ " + nf(x / 1e6, 1) + " mi" : "R$ " + nf(x / 1e3, 0) + " mil"
usd = x => x == null || x === 0 ? "–" : x >= 1e9 ? "US$ " + nf(x / 1e9, 2) + " bi" : x >= 1e6 ? "US$ " + nf(x / 1e6, 1) + " mi" : "US$ " + nf(x / 1e3, 0) + " mil"
AZUL = "#1F4E79"; LAR = "#E07A1F"; CINZA = "#B8C4CF"
cartao = (rotulo, valor, nota = "") => html`<div class="cartao"><div class="cartao-valor">${valor}</div><div class="cartao-rotulo">${rotulo}</div>${nota ? html`<div class="cartao-nota">${nota}</div>` : ""}</div>`

    <p class="rx-sub">Link direto: <code>${location.origin + location.pathname}?m=${m.id}</code></p>
  </div>
  <div class="rx-mapa">${Plot.plot({
    width: 440, height: 175, margin: 2,
    projection: {type: "mercator", domain: geo, inset: 4},
    marks: [
      Plot.geo(geo.features, {fill: d => d.properties.id_municipio === m.id ? LAR : "#DCE6F0", stroke: "white", strokeWidth: 0.3})
    ]})}</div>
</div>`

{
  let acc = 0;
  const seg = va.map(d => { const x1 = acc; acc += d.v / vatot; return {...d, x1, x2: acc, p: d.v / vatot}; });
  return Plot.plot({
    height: 110, marginLeft: 10, marginRight: 10, y: {axis: null}, x: {domain: [0, 1], label: "% do valor adicionado", tickFormat: d => nf(100 * d) + "%"},
    color: {domain: va.map(d => d.setor), range: ["#7FA66B", "#1F4E79", "#6C8EB2", "#B8C4CF"], legend: true},
    marks: [
      Plot.barX(seg, {x1: "x1", x2: "x2", y: () => "PIB", fill: "setor", insetRight: 1, title: d => `${d.setor}: ${nf(100 * d.p, 1)}%`}),
      Plot.text(seg, {x: d => (d.x1 + d.x2) / 2, y: () => "PIB", text: d => d.p > 0.07 ? nf(100 * d.p, 0) + "%" : "", fill: "white", fontWeight: "bold", fontSize: 13})
    ]
  });
}
